Basic CRUD

package main

import (
    "context"
    "fmt"
    "log"
    "time"
    ember "github.com/NepMods/ember"
)

type Product struct {
    ID        int64   `ember:"primaryKey;autoIncr"`
    Name      string
    Price     float64 `ember:"default:0"`
    Status    string  `ember:` + "`" + `default:"active"` + "`" + `
    CreatedAt time.Time
    UpdatedAt time.Time
}

func main() {
    ctx := context.Background()

    db, err := ember.Open(ember.Config{
        Driver: "sqlite3",
        Master: ":memory:",
    })
    if err != nil {
        log.Fatalf("failed to open db: %v", err)
    }
    defer db.Close()

    // Auto-migrate (create table from struct)
    _, err = db.Raw(`CREATE TABLE products (
        id INTEGER PRIMARY KEY AUTOINCREMENT,
        name TEXT NOT NULL,
        price REAL DEFAULT 0,
        status TEXT DEFAULT 'active',
        created_at TEXT,
        updated_at TEXT
    )`).Exec(ctx)
    if err != nil {
        log.Fatalf("failed to create table: %v", err)
    }

    // CREATE
    p := &Product{Name: "Widget", Price: 9.99}
    if err := db.Model().Create(ctx, p); err != nil {
        log.Fatalf("failed to create product: %v", err)
    }
    fmt.Printf("Created product #%d: %s\n", p.ID, p.Name)

    // READ (Find by PK)
    var found Product
    if err := db.Model().Find(ctx, &found, p.ID); err != nil {
        log.Fatalf("failed to find product: %v", err)
    }
    fmt.Printf("Found: %s ($%.2f)\n", found.Name, found.Price)

    // READ (First with condition)
    var first Product
    if err := db.Model().First(ctx, &first, func(b *ember.Builder) {
        b.Where("name", "LIKE", "%Widget%")
    }); err != nil {
        log.Fatalf("failed to find first product: %v", err)
    }
    fmt.Printf("First: %s\n", first.Name)

    // READ (All)
    var all []Product
    if err := db.Model().All(ctx, &all); err != nil {
        log.Fatalf("failed to list products: %v", err)
    }
    fmt.Printf("Total products: %d\n", len(all))

    // UPDATE
    found.Price = 14.99
    if err := db.Model().Save(ctx, &found); err != nil {
        log.Fatalf("failed to save product: %v", err)
    }

    // DELETE
    if err := db.Model().Delete(ctx, &found); err != nil {
        log.Fatalf("failed to delete product: %v", err)
    }
}

Transactions

// Simple transaction (auto-rollback on error)
err := db.Transaction(ctx, func(tx *ember.Tx) error {
    if err := tx.Table("accounts").Where("id", "=", 1).
        Update(ctx, map[string]interface{}{"balance": 100}); err != nil {
        return fmt.Errorf("update account 1: %w", err)
    }
    if err := tx.Table("accounts").Where("id", "=", 2).
        Update(ctx, map[string]interface{}{"balance": 200}); err != nil {
        return fmt.Errorf("update account 2: %w", err)
    }
    return nil
})
if err != nil {
    log.Fatalf("transaction failed: %v", err)
}

// Manual transaction with savepoints
tx, err := db.Begin(ctx, nil)
if err != nil {
    log.Fatalf("begin failed: %v", err)
}

// Use Builder within transaction
if _, err := tx.Table("orders").Insert(ctx, map[string]interface{}{
    "product_id": 1, "quantity": 2,
}); err != nil {
    tx.Rollback()
    log.Fatalf("insert failed: %v", err)
}

// Create a savepoint
if err := tx.Savepoint(ctx, "after_order"); err != nil {
    tx.Rollback()
    log.Fatalf("savepoint failed: %v", err)
}

// More operations
if _, err := tx.Table("inventory").Where("product_id", "=", 1).
    Decrement(ctx, "stock", 2); err != nil {
    tx.Rollback()
    log.Fatalf("decrement failed: %v", err)
}

// Rollback to savepoint on partial failure
if err := tx.RollbackTo(ctx, "after_order"); err != nil {
    tx.Rollback()
    log.Fatalf("rollback to savepoint failed: %v", err)
}

// Model API in transaction
var user User
if err := tx.Model().Find(ctx, &user, 1); err != nil {
    tx.Rollback()
    log.Fatalf("find in tx failed: %v", err)
}
user.Name = "Updated"
if err := tx.Model().Save(ctx, &user); err != nil {
    tx.Rollback()
    log.Fatalf("save in tx failed: %v", err)
}

// Raw queries in transaction
if _, err := tx.Raw("UPDATE accounts SET balance = balance - ? WHERE id = ?", 50, 1).Exec(ctx); err != nil {
    tx.Rollback()
    log.Fatalf("raw exec in tx failed: %v", err)
}

// Commit
if err := tx.Commit(); err != nil {
    log.Fatalf("commit failed: %v", err)
}

// Defer rollback on panic (for manual transactions)
tx2, err := db.Begin(ctx, nil)
if err != nil {
    log.Fatalf("begin failed: %v", err)
}
defer func() {
    if r := recover(); r != nil {
        tx2.Rollback()
        panic(r)
    }
}()

Migrations

// Define migrations
type CreateUsersTable struct{}
func (m *CreateUsersTable) Version() string { return "2024_01_01_create_users" }
func (m *CreateUsersTable) Up(s *ember.Schema) error {
    return s.Create("users", func(t *ember.Blueprint) {
        t.ID()
        t.String("name")
        t.String("email").Unique()
        t.String("status").Default("active")
        t.Timestamp("email_verified_at").Nullable()
        t.Timestamps()
        t.SoftDeletes()
    })
}
func (m *CreateUsersTable) Down(s *ember.Schema) error {
    return s.Drop("users")
}

type AddPhoneToUsers struct{}
func (m *AddPhoneToUsers) Version() string { return "2024_01_15_add_phone_to_users" }
func (m *AddPhoneToUsers) Up(s *ember.Schema) error {
    return s.Table("users", func(t *ember.Blueprint) {
        t.String("phone", 20).Nullable().After("email")
    })
}
func (m *AddPhoneToUsers) Down(s *ember.Schema) error {
    return s.Table("users", func(t *ember.Blueprint) {
        t.DropColumn("phone")
    })
}

// Run migrations
migrator := ember.NewMigrator(db, &CreateUsersTable{}, &AddPhoneToUsers{})

if err := migrator.Run(ctx); err != nil {
    log.Fatalf("migration run failed: %v", err)
}

// Status
statuses, err := migrator.Status(ctx)
if err != nil {
    log.Fatalf("migration status failed: %v", err)
}
for _, s := range statuses {
    fmt.Printf("%s: ran=%v\n", s.Version, s.Ran)
}

// Rollback last step
if err := migrator.Rollback(ctx, 1); err != nil {
    log.Fatalf("rollback failed: %v", err)
}

// Rollback all
if err := migrator.Rollback(ctx, 0); err != nil {
    log.Fatalf("rollback all failed: %v", err)
}

// Fresh (drop all + re-run)
if err := migrator.Fresh(ctx); err != nil {
    log.Fatalf("fresh failed: %v", err)
}

// Pending migrations
pending, err := migrator.Pending(ctx)
if err != nil {
    log.Fatalf("pending failed: %v", err)
}
fmt.Printf("%d pending migrations\n", len(pending))

// Custom migration table
migrator.SetTable("schema_migrations")

Relationships

// Define models with relationship tags
type User struct {
    ID        int64   `ember:"primaryKey;autoIncr"`
    Name      string
    Posts     []Post  `ember:"relation:hasMany;foreignKey:user_id;localKey:id"`
    Profile   Profile `ember:"relation:hasOne;foreignKey:user_id;localKey:id"`
}

type Post struct {
    ID        int64   `ember:"primaryKey;autoIncr"`
    UserID    int64   `ember:"column:user_id"`
    Title     string
    Body      string
    Author    User    `ember:"relation:belongsTo;foreignKey:user_id;ownerKey:id"`
    Tags      []Tag   `ember:"relation:belongsToMany;pivot:post_tags;pivotFK:post_id;pivotRK:tag_id"`
}

type Profile struct {
    ID     int64  `ember:"primaryKey;autoIncr"`
    UserID int64  `ember:"column:user_id"`
    Bio    string
}

type Tag struct {
    ID   int64  `ember:"primaryKey;autoIncr"`
    Name string
}

// Create related models
user := &User{Name: "Alice"}
if err := db.Model().Create(ctx, user); err != nil {
    log.Fatalf("create user failed: %v", err)
}

post := &Post{UserID: user.ID, Title: "Hello", Body: "World"}
if err := db.Model().Create(ctx, post); err != nil {
    log.Fatalf("create post failed: %v", err)
}

profile := &Profile{UserID: user.ID, Bio: "Gopher"}
if err := db.Model().Create(ctx, profile); err != nil {
    log.Fatalf("create profile failed: %v", err)
}

tag := &Tag{Name: "Go"}
if err := db.Model().Create(ctx, tag); err != nil {
    log.Fatalf("create tag failed: %v", err)
}

// Load relationships individually
var posts []Post
if err := db.Model().LoadHasMany(ctx, user, ember.HasMany{ForeignKey: "user_id"}, &posts); err != nil {

    log.Fatalf("load belongsTo failed: %v", err)
}

var tags []Tag
if err := db.Model().LoadBelongsToMany(ctx, post, ember.BelongsToMany{

Scopes

// Define reusable scopes
func PriceAbove(min float64) ember.Scope {
    return func(b *ember.Builder) *ember.Builder {
        return b.Where("price", ">=", min)
    }
}

func ActiveOnly(b *ember.Builder) *ember.Builder {
    return b.Where("status", "=", "active")
}

// Apply scopes to builder
if err := ember.ApplyScopes(db.Table("products"), ActiveOnly, PriceAbove(10.0)).Get(ctx, &products); err != nil {
    log.Fatalf("scope query failed: %v", err)
}

// Global scopes (automatically applied to all model queries)
db.ScopeRegistry().Add(&Product{}, ActiveOnly)

// Now all Product queries include WHERE status = 'active'
if err := db.Model().All(ctx, &products); err != nil {  // includes scope
    log.Fatalf("all with scope failed: %v", err)
}
if err := db.Model().Find(ctx, &product, 1); err != nil {  // includes scope
    log.Fatalf("find with scope failed: %v", err)
}

// Built-in scope helpers
if err := ember.ApplyScopes(db.Table("products"), ember.SoftDeleteScope).Get(ctx, &products); err != nil {
    log.Fatalf("soft delete scope failed: %v", err)
}

// Include soft-deleted records
if err := ember.ApplyScopes(db.Table("products"), ember.WithTrashedScope()).Get(ctx, &products); err != nil {
    log.Fatalf("with trashed scope failed: %v", err)
}

// Only soft-deleted records
if err := ember.ApplyScopes(db.Table("products"), ember.OnlyTrashedScope()).Get(ctx, &products); err != nil {
    log.Fatalf("only trashed scope failed: %v", err)
}

Events & Hooks

// Model hooks (interface-based)
type User struct {
    ID   int64  `ember:"primaryKey;autoIncr"`
    Name string
    Age  int
}

func (u *User) BeforeCreate(ctx context.Context) error {
    if u.Name == "" {
        return fmt.Errorf("name is required")
    }
    return nil
}

func (u *User) AfterCreate(ctx context.Context) error {
    log.Printf("User %d created: %s", u.ID, u.Name)
    return nil
}

// Observer pattern
type AuditObserver struct {
    ember.BaseObserver // provides no-op defaults
}

func (o *AuditObserver) Created(ctx context.Context, model interface{}) error {
    log.Printf("[AUDIT] Created: %T %+v", model, model)
    return nil
}

func (o *AuditObserver) Deleted(ctx context.Context, model interface{}) error {
    log.Printf("[AUDIT] Deleted: %T %+v", model, model)
    return nil
}

// ObserverFuncs (functional style)
funcTimeObserver := &ember.ObserverFuncs{
    SavedFunc: func(ctx context.Context, m interface{}) error {
        log.Printf("Saved: %+v", m)
        return nil
    },
    DeletedFunc: func(ctx context.Context, m interface{}) error {
        log.Printf("Deleted: %+v", m)
        return nil
    },
}

// Register
dispatcher := ember.NewEventDispatcher()
dispatcher.Observe(&User{}, &AuditObserver{})
dispatcher.Observe(&User{}, funcTimeObserver)
dispatcher.ObserveAll(&AuditObserver{}) // global observer
db.SetEventDispatcher(dispatcher)

// Events fire automatically on model operations
if err := db.Model().Create(ctx, &User{Name: "Alice"}); err != nil {
    log.Fatalf("create with events failed: %v", err)
}
if err := db.Model().Delete(ctx, &user); err != nil {
    log.Fatalf("delete with events failed: %v", err)
}

Macros

// Per-instance macro
builder := db.Table("users")
builder.Macro("whereActive", func(b *ember.Builder, args ...interface{}) interface{} {
    return b.Where("status", "=", "active")
})

result, err := builder.Call("whereActive")
if err != nil {
    log.Fatalf("macro call failed: %v", err)
}
if b, ok := result.(*ember.Builder); ok {
    if err := b.Get(ctx, &users); err != nil {
        log.Fatalf("macro get failed: %v", err)
    }
}

// Macro that returns a value
builder.Macro("countWhere", func(b *ember.Builder, args ...interface{}) interface{} {
    col := args[0].(string)
    op := args[1].(string)
    val := args[2]
    count, err := b.Where(col, op, val).Count(context.Background())
    if err != nil {
        return err
    }
    return count
})

result, err := builder.Call("countWhere", "age", ">", 18)
if err != nil {
    log.Fatalf("count macro failed: %v", err)
}
if count, ok := result.(int64); ok {
    fmt.Printf("Count: %d\n", count)
}

// Global macro (available on all builders)
ember.AddBuilderMacro("paginated", func(b *ember.Builder, args ...interface{}) interface{} {
    page := args[0].(int)
    perPage := args[1].(int)
    return b.ForPage(page, perPage)
})

result, err := db.Table("users").Call("paginated", 1, 20)
if err != nil {
    log.Fatalf("global macro failed: %v", err)
}
if b, ok := result.(*ember.Builder); ok {
    if err := b.Get(ctx, &users); err != nil {
        log.Fatalf("global macro get failed: %v", err)
    }
}

Real-World: Blog API

A complete example combining models, migrations, relationships, eager loading, pagination, serialization, events, and transactions — simulating a blog REST API backend.

package main

import (
    "context"
    "fmt"
    "log"
    "time"
    ember "github.com/NepMods/ember"
)

// ─── Models ─────────────────────────────────────

type User struct {
    ID        int64     `ember:"primaryKey;autoIncr"`
    Name      string    `ember:"column:name"`
    Email     string    `ember:"unique"`
    Status    string    `ember:` + "`" + `default:"active"` + "`" + `
    Posts     []Post    `ember:"relation:hasMany;foreignKey:user_id;localKey:id"`
    CreatedAt time.Time
    UpdatedAt time.Time
}

type Post struct {
    ID        int64     `ember:"primaryKey;autoIncr"`
    UserID    int64     `ember:"column:user_id"`
    Title     string
    Body      string    `ember:"cast:json"`
    Status    string    `ember:` + "`" + `default:"draft"` + "`" + `
    Tags      []Tag     `ember:"relation:belongsToMany;pivot:post_tags;pivotFK:post_id;pivotRK:tag_id"`
    Author    User      `ember:"relation:belongsTo;foreignKey:user_id;ownerKey:id"`
    CreatedAt time.Time
    UpdatedAt time.Time
    DeletedAt *time.Time `ember:"nullable"`
}

type Tag struct {
    ID        int64     `ember:"primaryKey;autoIncr"`
    Name      string    `ember:"unique"`
    CreatedAt time.Time
}

// ─── Scaffold ───────────────────────────────────

func createTables(ctx context.Context, db *ember.DB) error {
    schema := ember.NewSchema(db)

    if err := schema.Create("users", func(t *ember.Blueprint) {
        t.ID()
        t.String("name")
        t.String("email").Unique()
        t.String("status").Default("active")
        t.Timestamps()
    }); err != nil {
        return fmt.Errorf("create users: %w", err)
    }

    if err := schema.Create("posts", func(t *ember.Blueprint) {
        t.ID()
        t.UnsignedBigInteger("user_id")
        t.String("title")
        t.Text("body")
        t.String("status").Default("draft")
        t.Timestamps()
        t.SoftDeletes()
        t.Foreign("user_id").References("id").On("users").CascadeOnDelete()
    }); err != nil {
        return fmt.Errorf("create posts: %w", err)
    }

    if err := schema.Create("tags", func(t *ember.Blueprint) {
        t.ID()
        t.String("name").Unique()
        t.Timestamps()
    }); err != nil {
        return fmt.Errorf("create tags: %w", err)
    }

    if err := schema.Create("post_tags", func(t *ember.Blueprint) {
        t.UnsignedBigInteger("post_id")
        t.UnsignedBigInteger("tag_id")
        t.Primary("post_id", "tag_id")
        t.Foreign("post_id").References("id").On("posts").CascadeOnDelete()
        t.Foreign("tag_id").References("id").On("tags").CascadeOnDelete()
    }); err != nil {
        return fmt.Errorf("create post_tags: %w", err)
    }

    return nil
}

// ─── Seed ───────────────────────────────────────

func seedData(ctx context.Context, db *ember.DB) error {
    users := []User{
        {Name: "Alice", Email: "alice@example.com"},
        {Name: "Bob", Email: "bob@example.com"},
    }
    for i := range users {
        if err := db.Model().Create(ctx, &users[i]); err != nil {
            return fmt.Errorf("create user %d: %w", i, err)
        }
    }

    tags := []Tag{
        {Name: "go"}, {Name: "tutorial"}, {Name: "news"},
    }
    for i := range tags {
        if err := db.Model().Create(ctx, &tags[i]); err != nil {
            return fmt.Errorf("create tag %d: %w", i, err)
        }
    }

    posts := []Post{
        {UserID: users[0].ID, Title: "Getting Started with Go", Body: "Go is a statically typed...", Status: "published"},
        {UserID: users[0].ID, Title: "Advanced ORM Patterns", Body: "When building complex apps...", Status: "draft"},
        {UserID: users[1].ID, Title: "Go Modules Guide", Body: "Go modules are the official...", Status: "published"},
    }
    for i := range posts {
        if err := db.Model().Create(ctx, &posts[i]); err != nil {
            return fmt.Errorf("create post %d: %w", i, err)
        }
    }

    // Attach tags to posts
    type postTag struct {
        PostID int64 `ember:"column:post_id"`
        TagID  int64 `ember:"column:tag_id"`
    }
    links := []postTag{
        {PostID: posts[0].ID, TagID: tags[0].ID},
        {PostID: posts[0].ID, TagID: tags[1].ID},
        {PostID: posts[1].ID, TagID: tags[0].ID},
        {PostID: posts[2].ID, TagID: tags[2].ID},
        {PostID: posts[2].ID, TagID: tags[0].ID},
    }
    for _, l := range links {
        m := map[string]interface{}{"post_id": l.PostID, "tag_id": l.TagID}
        if _, err := db.Table("post_tags").Insert(ctx, m); err != nil {
            return fmt.Errorf("link post %d to tag %d: %w", l.PostID, l.TagID, err)
        }
    }

    return nil
}

// ─── Queries ────────────────────────────────────

func listPublishedPosts(ctx context.Context, db *ember.DB, page, perPage int) error {
    var posts []Post
    paginator, err := db.Model().With("Author", "Tags").Paginate(ctx, &posts, page, perPage,
        func(b *ember.Builder) {
            b.Where("status", "=", "published").OrderBy("created_at", "DESC")
        },
    )
    if err != nil {
        return fmt.Errorf("paginate posts: %w", err)
    }

    fmt.Printf("Page %d of %d (total: %d posts)\n",
        paginator.CurrentPage, paginator.LastPage, paginator.Total)

    for _, p := range posts {
        tagNames := make([]string, len(p.Tags))
        for i, t := range p.Tags {
            tagNames[i] = t.Name
        }
        fmt.Printf("  #%d [%s] %s by %s — Tags: %v\n",
            p.ID, p.Status, p.Title, p.Author.Name, tagNames)
    }
    return nil
}

func createPostWithTransaction(ctx context.Context, db *ember.DB, userID int64, title, body string, tagIDs []int64) (*Post, error) {
    var post *Post

    err := db.Transaction(ctx, func(tx *ember.Tx) error {
        post = &Post{UserID: userID, Title: title, Body: body, Status: "draft"}
        if err := tx.Model().Create(ctx, post); err != nil {
            return fmt.Errorf("create post: %w", err)
        }

        for _, tagID := range tagIDs {
            m := map[string]interface{}{"post_id": post.ID, "tag_id": tagID}
            if _, err := tx.Table("post_tags").Insert(ctx, m); err != nil {
                return fmt.Errorf("attach tag %d: %w", tagID, err)
            }
        }
        return nil
    })
    if err != nil {
        return nil, err
    }

    return post, nil
}

func softDeleteAndRestore(ctx context.Context, db *ember.DB, postID int64) error {
    var post Post
    if err := db.Model().Find(ctx, &post, postID); err != nil {
        return fmt.Errorf("find post: %w", err)
    }

    if err := db.Model().Delete(ctx, &post); err != nil {
        return fmt.Errorf("soft delete: %w", err)
    }
    fmt.Printf("Soft-deleted post #%d at %v\n", post.ID, post.DeletedAt)

    var trashed []Post
    if err := ember.ApplyScopes(db.Table("posts"), ember.OnlyTrashedScope()).Get(ctx, &trashed); err != nil {
        return fmt.Errorf("find trashed: %w", err)
    }
    fmt.Printf("Trashed posts: %d\n", len(trashed))

    if err := db.Model().Restore(ctx, &post); err != nil {
        return fmt.Errorf("restore: %w", err)
    }
    fmt.Printf("Restored post #%d\n", post.ID)
    return nil
}

func searchPosts(ctx context.Context, db *ember.DB, query string) error {
    var results []Post
    if err := db.Table("posts").
        Where("title", "LIKE", "%"+query+"%").
        OrWhere("body", "LIKE", "%"+query+"%").
        WhereNull("deleted_at").
        OrderBy("created_at", "DESC").
        Limit(20).
        Get(ctx, &results); err != nil {
        return fmt.Errorf("search: %w", err)
    }
    fmt.Printf("Found %d posts matching %q\n", len(results), query)
    return nil
}

// ─── Main ───────────────────────────────────────

func main() {
    ctx := context.Background()

    db, err := ember.Open(ember.Config{
        Driver: "sqlite3",
        Master: ":memory:",
    })
    if err != nil {
        log.Fatalf("open: %v", err)
    }
    defer db.Close()

    if err := createTables(ctx, db); err != nil {
        log.Fatalf("migrate: %v", err)
    }
    fmt.Println("Tables created.")

    if err := seedData(ctx, db); err != nil {
        log.Fatalf("seed: %v", err)
    }
    fmt.Println("Seed data inserted.")

    // List published posts with eager loading + pagination
    if err := listPublishedPosts(ctx, db, 1, 10); err != nil {
        log.Fatalf("list: %v", err)
    }

    // Create a post with tags in a transaction
    post, err := createPostWithTransaction(ctx, db, 1, "New Post", "Content here", []int64{1, 2})
    if err != nil {
        log.Fatalf("create post: %v", err)
    }
    fmt.Printf("Created post #%d in transaction\n", post.ID)

    // Soft delete and restore
    if err := softDeleteAndRestore(ctx, db, post.ID); err != nil {
        log.Fatalf("soft delete/restore: %v", err)
    }

    // Full-text search
    if err := searchPosts(ctx, db, "Go"); err != nil {
        log.Fatalf("search: %v", err)
    }

    // Serialize a user with posts to JSON
    var alice User
    if err := db.Model().With("Posts").Find(ctx, &alice, 1); err != nil {
        log.Fatalf("find alice: %v", err)
    }
    json, err := ember.ToJSON(alice)
    if err != nil {
        log.Fatalf("serialize: %v", err)
    }
    fmt.Printf("Alice JSON: %s\n", string(json))
}
